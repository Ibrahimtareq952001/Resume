\usepackage{enumitem}

% An entry with no description: unlike \cventry with an empty 5th argument,
% it doesn't leave a blank row underneath.
% Usage: \cvshortentry{<position>}{<title>}{<location>}{<date>}
\newcommand*{\cvshortentry}[4]{%
  \vspace{-2.0mm}%
  \setlength\tabcolsep{0pt}%
  \setlength{\extrarowheight}{0pt}%
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - 4.5cm} R{4.5cm}}
    \entrytitlestyle{#2} & \entrylocationstyle{#3} \\
    \entrypositionstyle{#1} & \entrydatestyle{#4} \\
  \end{tabular*}%
  \vspace{-4.0mm}% same as the end of cvitems, so every section ends alike
}
